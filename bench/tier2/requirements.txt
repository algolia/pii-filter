# Bench-only dependencies for bench/tier2/ (Tier-2 PII NER latency + accuracy receipts).
# These are NOT toolkit dependencies -- keep them out of the package pyproject. Install
# into a dedicated venv. Torch must be the CPU wheel:
#   python -m venv venv && . venv/bin/activate
#   pip install torch --index-url https://download.pytorch.org/whl/cpu
#   pip install -r requirements.txt
#
# Versions below are what these first measurements were taken with (2026-07-15).
torch  # CPU wheel, see index-url above; measured with 2.13.0+cpu
gliner2[local]==1.3.2   # GLiNER2-PII (fastino/gliner2-privacy-filter-PII-multi)
gliner==0.2.27          # knowledgator/gliner-pii-edge (see NOTE below)
onnxruntime
huggingface_hub
pandas                  # dataset generator: read INSEE prenoms parquet
pyarrow                 # parquet engine
psutil                  # (optional) memory; bench uses stdlib resource by default
matplotlib              # charts (docs/assets/render_tier2_*.py)
numpy

# NOTE: gliner-pii-edge-v1.0 uses a jhu-clsp/ettin-encoder-32m (ModernBERT) encoder that
# gliner 0.2.27 does NOT run correctly -- the vendor's own model-card example returns zero
# detections at threshold 0.3, across the PyTorch and ONNX (quint8/fp32) paths (fp16 fails
# to load with an onnxruntime fp16 type error). Its accuracy is therefore unmeasurable in
# this environment; only its forward-pass latency/footprint is reported. See the report's
